```sql
-- Crie uma tabela para armazenar os dados dos produtos
CREATE TABLE produtos (
    id SERIAL PRIMARY KEY,
    titulo VARCHAR(100) NOT NULL,
    autor VARCHAR(100) NOT NULL,
    categoria VARCHAR(100) NOT NULL,
    tipo VARCHAR(10) NOT NULL,
    preco DECIMAL(10,2) NOT NULL,
    quantidade_em_estoque INTEGER NOT NULL,
    formato_ebook VARCHAR(20),
    numero_de_paginas INTEGER,
    idioma VARCHAR(20),
    editora VARCHAR(50),
    ano_de_publicacao INTEGER,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Crie uma tabela para armazenar as avaliações dos clientes
CREATE TABLE avaliacoes (
    id SERIAL PRIMARY KEY,
    produto_id INTEGER NOT NULL,
    avaliacao DECIMAL(5,2) NOT NULL,
    comentario TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (produto_id) REFERENCES produtos (id)
);

-- Crie uma tabela para armazenar os pedidos dos clientes
CREATE TABLE pedidos (
    id SERIAL PRIMARY KEY,
    produto_id INTEGER NOT NULL,
    quantidade INTEGER NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (produto_id) REFERENCES produtos (id)
);

-- Crie uma tabela para armazenar as promoções e descontos
CREATE TABLE promocoes (
    id SERIAL PRIMARY KEY,
    produto_id INTEGER NOT NULL,
    preco DECIMAL(10,2) NOT NULL,
    forma_de_desconto VARCHAR(20) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (produto_id) REFERENCES produtos (id)
);

-- Crie uma tabela para armazenar as capas e imagens dos produtos
CREATE TABLE capas (
    id SERIAL PRIMARY KEY,
    produto_id INTEGER NOT NULL,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (produto_id) REFERENCES produtos (id)
);

-- Crie uma tabela para armazenar a data de cadastro e última atualização
CREATE TABLE registros (
    id SERIAL PRIMARY KEY,

);
```
