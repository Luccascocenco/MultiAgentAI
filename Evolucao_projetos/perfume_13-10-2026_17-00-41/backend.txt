**Backend Python para Esquema do Banco de Dados**

```python
from flask import Flask, jsonify, request
import sqlite3
import os

# Configuração do aplicativo
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
DB_PATH = os.path.join(BASE_DIR, "database.db")
SCHEMA_PATH = os.path.join(BASE_DIR, "database", "schema.sql")

# Criação do aplicativo
app = Flask(__name__, static_folder="static", static_url_path="")


# Definição da função de conexão com o banco de dados
def get_db_connection():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


# Definição da função de inicialização do banco de dados
def inicializar_banco():
    # Criação do banco de dados a partir do esquema
    if os.path.exists(DB_PATH) and os.path.getsize(DB_PATH) > 0:
        return

    conn = get_db_connection()
    with open(SCHEMA_PATH, encoding="utf-8") as arquivo:
        conn.executescript(arquivo.read())
    conn.commit()
    conn.close()


# Definição das rotas da API
@app.route("/produtos")
def listar_produtos():
    conn = get_db_connection()
    linhas = conn.execute("SELECT id, nome, imagem, descricao, categoria, volume, preco FROM produtos").fetchall()
    conn.close()
    return jsonify([dict(linha) for linha in linhas])


@app.route("/produtos/<int:item_id>")
def obter_produto(item_id):
    conn = get_db_connection()
    linha = conn.execute(
        "SELECT id, nome, imagem, descricao, categoria, volume, preco FROM produtos WHERE id = ?",
        (item_id,)
    ).fetchone()
    conn.close()
    if linha is None:
        return jsonify({"erro": "Produto não encontrado"}), 404
    return jsonify(dict(linha))


@app.route("/carrinho", methods=["POST"])
def atualizar_carrinho():
    dados = request.get_json()
    conn = get_db_connection()
    conn.execute("UPDATE carrinhos SET quantidade = ? WHERE id_usuario = ? AND id_produto = ?", (dados["quantidade"], dados["id_usuario"], dados["id_produto"]))
    conn.commit()
    conn.close()
    return jsonify({" mensagem": "Atualizado com sucesso" })


@app.route("/pedidos", methods=["POST"])
def criar_pedido():

# Inicialização do banco de dados
inicializar_banco()


# Início do servidor
if __name__ == "__main__":
    porta = int(os.environ.get("PORT", 5000))
    app.run(debug=False, port=porta)
```

**Esquema do Banco de Dados**

```sql
CREATE TABLE produtos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    imagem TEXT,
    descricao TEXT,
    categoria TEXT,
    volume TEXT,
    preco REAL NOT NULL,
    quantidade INTEGER NOT NULL
);

CREATE TABLE pedidos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    id_usuario INTEGER NOT NULL,
    id_produto INTEGER NOT NULL,
    quantidade INTEGER NOT NULL,
    data DATE NOT NULL,
    FOREIGN KEY (id_usuario) REFERENCES usuarios (id),
    FOREIGN KEY (id_produto) REFERENCES produtos (id)
);

CREATE TABLE usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    email TEXT UNIQUE NOT NULL,
    senha TEXT NOT NULL
);

CREATE TABLE carrinhos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    id_usuario INTEGER NOT NULL,
    id_produto INTEGER NOT NULL,
    quantidade INTEGER NOT NULL,
    data DATE NOT NULL,
    FOREIGN KEY (id_usuario) REFERENCES usuarios (id),
    FOREIGN KEY (id_produto) REFERENCES produtos (id)
);
```
