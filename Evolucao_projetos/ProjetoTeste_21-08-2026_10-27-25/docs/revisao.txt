```javascript
// src/services/autenticacao.js
const express = require('express');
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');

const autenticacao = async (req, res) => {
  const { email, senha } = req.body;

  const vendedor = await Vendedor.findOne({ email });

  if (!vendedor) {
    return res.status(401).send({ mensagem: 'Email ou senha incorretos' });
  }

  const compararSenhas = bcrypt.compareSync(senha, vendedor.senha);

  if (!compararSenhas) {
    return res.status(401).send({ mensagem: 'Email ou senha incorretos' });
  }

  const token = jwt.sign({ id: vendedor._id }, 'segredo');

  return res.send({ token });
};

module.exports = autenticacao;

// src/services/gerenciamentoDeProdutos.js
const express = require('express');
const Livro = require('../models/livro');

const router = express.Router();

const adicionarProduto = async (req, res) => {
  const { titulo, autor, isbn, preco, estoque } = req.body;
  const livro = new Livro({ titulo, autor, isbn, preco, estoque });
  await livro.save();

  return res.send({ mensagem: 'Produto adicionado com sucesso' });
};

module.exports = adicionarProduto;

// src/services/gerenciamentoDePedidos.js
const express = require('express');
const Pedido = require('../models/pedido');

const router = express.Router();

const criarPedido = async (req, res) => {
  const { vendedorId, clienteId, data, total } = req.body;
  const pedido = new Pedido({ vendedor: vendedorId, cliente: clienteId, data, total });
  await pedido.save();

  return res.send({ mensagem: 'Pedido criado com sucesso' });
};

module.exports = criarPedido;

// src/services/gerenciamentoDeContas.js
const express = require('express');
const Conta = require('../models/conta');

const router = express.Router();

const gerenciarSaldo = async (req, res) => {
  const { vendedorId, clienteId, saldo } = req.body;
  const conta = Conta.findOne({ vendedor: vendedorId, cliente: clienteId });

  if (!conta) {
    const novaConta = new Conta({ vendedor: vendedorId, cliente: clienteId, saldo });
    await novaConta.save();

    return res.send({ mensagem: 'Conta criada com sucesso' });
  }

  conta.saldo = saldo;
  await conta.save();

  return res.send({ mensagem: 'Saldo atualizado com sucesso' });
};

module.exports = gerenciarSaldo;

// src/services/gerenciamentoDeRecursos.js
const express = require('express');
const Livro = require('../models/livro');

const router = express.Router();

const gerenciarEstoque = async (req, res) => {
  const { livroId, estoque } = req.body;
  const livro = await Livro.findById(livroId);

  livro.estoque = estoque;
  await livro.save();

  return res.send({ mensagem: 'Estoque atualizado com sucesso' });
};

module.exports = gerenciarEstoque;

// src/constructors/aplicacao.js
const express = require('express');
const router = express.Router();
const autenticacao = require('../services/autenticacao');
const gerenciamentoDeProdutos = require('../services/gerenciamentoDeProdutos');
const gerenciamentoDePedidos = require('../services/gerenciamentoDePedidos');
const gerenciamentoDeContas = require('../services/gerenciamentoDeContas');
const gerenciamentoDeRecursos = require('../services/gerenciamentoDeRecursos');

router.post('/autenticacao', autenticacao);
router.post('/produtos', gerenciamentoDeProdutos.adicionarProduto);
router.post('/pedidos', gerenciamentoDePedidos.criarPedido);
router.post('/contas', gerenciamentoDeContas.gerenciarSaldo);
router.post('/recursos', gerenciamentoDeRecursos.gerenciarEstoque);

module.exports = router;

// app.js
const express = require('express');
const app = express();
const port = 3000;

const router = require('./src/constructors/aplicacao');
app.use('/api', router);

app.listen(port, () => {
  console.log(`Servidor rodando na porta ${port}`);
});
```

**Banco de Dados**
```sql
CREATE DATABASE lojaonline;

USE lojaonline;

CREATE TABLE categorias (
    id INTEGER PRIMARY KEY,
    nome TEXT NOT NULL
);

CREATE TABLE livros (
    id INTEGER PRIMARY KEY,
    titulo TEXT NOT NULL,
    autor TEXT NOT NULL,
    isbn TEXT NOT NULL,
    categoria_id INTEGER NOT NULL,
    FOREIGN KEY (categoria_id) REFERENCES categorias(id)
);

CREATE TABLE vendedores (
    id INTEGER PRIMARY KEY,
    nome TEXT NOT NULL,
    email TEXT NOT NULL,
    senha TEXT NOT NULL
);

CREATE TABLE clientes (
    id INTEGER PRIMARY KEY,
    nome TEXT NOT NULL,
    email TEXT NOT NULL,
    senha TEXT NOT NULL
);

CREATE TABLE pedidos (
    id INTEGER PRIMARY KEY,
    data TEXT NOT NULL,
    total REAL NOT NULL,
    status TEXT NOT NULL,
    vendedor_id INTEGER NOT NULL,
    cliente_id INTEGER NOT NULL,
    FOREIGN KEY (vendedor_id) REFERENCES vendedores(id),
    FOREIGN KEY (cliente_id) REFERENCES clientes(id)
);

CREATE TABLE produtos (
    id INTEGER PRIMARY KEY,
    nome TEXT NOT NULL,
    descricao TEXT NOT NULL,
    preco REAL NOT NULL,
    imagem TEXT NOT NULL,
    vendedor_id INTEGER NOT NULL,
    FOREIGN KEY (vendedor_id) REFERENCES vendedores(id)
);

CREATE TABLE pedidos_itens (
    id INTEGER PRIMARY KEY,
    pedido_id INTEGER NOT NULL,
    produto_id INTEGER NOT NULL,
    quantidade INTEGER NOT NULL,
    FOREIGN KEY (pedido_id) REFERENCES pedidos(id),
    FOREIGN KEY (produto_id) REFERENCES produtos(id)
);

CREATE TABLE contas (
    id INTEGER PRIMARY KEY,
    tipo TEXT NOT NULL,
    saldo REAL NOT NULL,
    vendedor_id INTEGER NOT NULL,
    FOREIGN KEY (vendedor_id) REFERENCES vendedores(id)
);

CREATE TABLE contas_itens (
    id INTEGER PRIMARY KEY,
    conta_id INTEGER NOT NULL,
    data TEXT NOT NULL,
    tipo TEXT NOT NULL,
    valor REAL NOT NULL,
    FOREIGN KEY (conta_id) REFERENCES contas(id)
);
```

**Procedimentos SQL**
```sql
CREATE PROCEDURE sp_inserirLivro(
    @titulo TEXT,
    @autor TEXT,
    @isbn TEXT,
    @categoria_id INTEGER
) AS
BEGIN
    INSERT INTO livros (titulo, autor, isbn, categoria_id) VALUES (@titulo, @autor, @isbn, @categoria_id);
END;

CREATE PROCEDURE sp_inserirVendedor(
    @nome TEXT,
    @email TEXT,
    @senha TEXT
) AS
BEGIN
    INSERT INTO vendedores (nome, email, senha) VALUES (@nome, @email, @senha);
END;

CREATE PROCEDURE sp_inserirCliente(
    @nome TEXT,
    @email TEXT,
    @senha TEXT
) AS
BEGIN
    INSERT INTO clientes (nome, email, senha) VALUES (@nome, @email, @senha);
END;

CREATE PROCEDURE sp_inserirPedido(
    @data TEXT,
    @total REAL,
    @status TEXT,
    @vendedor_id INTEGER,
    @cliente_id INTEGER
) AS
BEGIN
    INSERT INTO pedidos (data, total, status, vendedor_id, cliente_id) VALUES (@data, @total, @status, @vendedor_id, @cliente_id);
END;

CREATE PROCEDURE sp_inserirProduto(
    @nome TEXT,
    @descricao TEXT,
    @preco REAL,
    @imagem TEXT,
    @vendedor_id INTEGER
) AS
BEGIN
    INSERT INTO produtos (nome, descricao, preco, imagem, vendedor_id) VALUES (@nome, @descricao, @preco, @imagem, @vendedor_id);
END;

CREATE PROCEDURE sp_inserirPedidoItem(
    @pedido_id INTEGER,
    @produto_id INTEGER,
    @quantidade INTEGER
) AS
BEGIN
    INSERT INTO pedidos_itens (pedido_id, produto_id, quantidade) VALUES (@pedido_id, @produto_id, @quantidade);
END;

CREATE PROCEDURE sp_inserirConta(
    @tipo TEXT,
    @saldo REAL,
    @vendedor_id INTEGER
) AS
BEGIN
    INSERT INTO contas (tipo, saldo, vendedor_id) VALUES (@tipo, @saldo, @vendedor_id);
END;

CREATE PROCEDURE sp_inserirContaItem(
    @conta_id INTEGER,
    @data TEXT,
    @tipo TEXT,
    @valor REAL
) AS
BEGIN
    INSERT INTO contas_itens (conta_id, data, tipo, valor) VALUES (@conta_id, @data, @tipo, @valor);
END;
```
