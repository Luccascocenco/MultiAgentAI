**script.js:**
```javascript
const produtos = [
  { id: 1, nome: 'Boneca Sofia', preco: 89.90, quantidade: 1 },
  { id: 2, nome: 'Boneca Alice', preco: 119.90, quantidade: 1 },
  { id: 3, nome: 'Boneca Luna', preco: 149.90, quantidade: 1 },
  { id: 4, nome: 'Boneca Valentina', preco: 99.90, quantidade: 1 },
  { id: 5, nome: 'Boneca Bella', preco: 129.90, quantidade: 1 },
];

const carrinho = document.getElementById('produtos-carrinho');
const subtotal = document.getElementById('subtotal');
const total = document.getElementById('total');
const removerProduto = document.getElementById('remover-produto');
const finalizarCompra = document.getElementById('finalizar-compra');

let produtosCarrinho = [];

// Função para adicionar um produto ao carrinho
function adicionarProdutoAoCarrinho(produto) {
  const indice = produtosCarrinho.findIndex((item) => item.id === produto.id);
  if (indice !== -1) {
    produtosCarrinho[indice].quantidade++;
  } else {
    produtosCarrinho.push(produto);
  }
  atualizarSubtotal();
  atualizarTotal();
}

// Função para remover um produto do carrinho
function removerProdutoDoCarrinho(produto) {
  const indice = produtosCarrinho.findIndex((item) => item.id === produto.id);
  if (indice !== -1) {
    produtosCarrinho[indice].quantidade--;
    if (produtosCarrinho[indice].quantidade === 0) {
      produtosCarrinho.splice(indice, 1);
    }
  }
  atualizarSubtotal();
  atualizarTotal();
}

// Função para atualizar subtotal e total
function atualizarSubtotal() {
  let subtotal = 0;
  produtosCarrinho.forEach((produto) => {
    subtotal += produto.preco * produto.quantidade;
  });
  document.getElementById('subtotal').textContent = `R$ ${subtotal.toFixed(2)}`;
}

function atualizarTotal() {
  let total = 0;
  produtosCarrinho.forEach((produto) => {
    total += produto.preco * produto.quantidade;
  });
  document.getElementById('total').textContent = `R$ ${total.toFixed(2)}`;
}

// Evento para adicionar produto ao carrinho
removerProduto.addEventListener('click', () => {
  produtosCarrinho.forEach((produto) => {
    if (produto.quantidade > 1) {
      produto.quantidade--;
    } else {
      produtosCarrinho.splice(produtosCarrinho.indexOf(produto), 1);
    }
  });
  atualizarSubtotal();
  atualizarTotal();
});

// Evento para finalizar compra
finalizarCompra.addEventListener('click', () => {
  // Lógica para finalizar compra
  console.log('Compra finalizada!');
});

// Evento para adicionar produto ao carrinho
carrinho.addEventListener('click', (event) => {
  if (event.target.classList.contains('produto')) {
    const produto = event.target.dataset.produto;
    adicionarProdutoAoCarrinho(produto);
  }
});
```
**backend.py:**
```python
from flask import Flask, request, jsonify
import sqlite3

app = Flask(__name__, static_folder="static", static_url_path="")

# Conexão com o banco de dados
conn = sqlite3.connect('database.db')
cursor = conn.cursor()

# Criação das tabelas
cursor.execute('''
  CREATE TABLE IF NOT EXISTS produtos (
    id INTEGER PRIMARY KEY,
    nome TEXT NOT NULL,
    preco REAL NOT NULL,
    quantidade INTEGER DEFAULT 1
  );
''')

# Insere os produtos
products = [
  ('Boneca Sofia', 89.90, 1),
  ('Boneca Alice', 119.90, 1),
  ('Boneca Luna', 149.90, 1),
  ('Boneca Valentina', 99.90, 1),
  ('Boneca Bella', 129.90, 1),
]
cursor.executemany('INSERT INTO produtos (nome, preco, quantidade) VALUES (?, ?, ?)', products)

# Fechamento da conexão
conn.commit()
conn.close()

@app.route('/')
def index():
  return 'Bem-vindo ao site Pink Dolls!'

@app.route('/api/produtos', methods=["GET"])
def listar_produtos():
  conn = get_db_connection()
  linhas = conn.execute("SELECT id, nome, preco, quantidade FROM produtos").fetchall()
  conn.close()
  return jsonify([dict(linha) for linha in linhas])

@app.route('/api/produtos/<int:item_id>', methods=["GET"])
def obter_produto(item_id):
  conn = get_db_connection()
  linha = conn.execute("SELECT id, nome, preco, quantidade FROM produtos WHERE id = ?", (item_id,)).fetchone()
  conn.close()
  if linha is None:
    return jsonify({"erro": "Produto não encontrado"}), 404
  return jsonify(dict(linha))

@app.route('/api/produtos', methods=["POST"])
def criar_produto():
  dados = request.get_json()
  conn = get_db_connection()
  cursor = conn.execute("INSERT INTO produtos (nome, preco, quantidade) VALUES (?, ?, ?)", (dados["nome"], dados["preco"], dados["quantidade"]))
  conn.commit()
  novo_id = cursor.lastrowid
  conn.close()
  return jsonify({"id": novo_id}), 201

@app.route('/api/produtos/<int:item_id>', methods=["DELETE"])
def remover_produto(item_id):
  conn = get_db_connection()
  conn.execute("DELETE FROM produtos WHERE id = ?", (item_id,))
  conn.commit()
  conn.close()
  return jsonify({"mensagem": "Produto removido com sucesso"})

def get_db_connection():
  conn = sqlite3.connect('database.db')
  conn.row_factory = sqlite3.Row
  return conn
```
