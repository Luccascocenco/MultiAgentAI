```javascript
const products = [
  {
    id: 1,
    nome: "Boneca Sofia",
    imagem: "img/sofia.jpg",
    descrição: "Boneca com vestido rosa e cabelos longos.",
    preço: 89.90
  },
  {
    id: 2,
    nome: "Boneca Alice",
    imagem: "img/alice.jpg",
    descrição: "Boneca fashion com acessórios especiais.",
    preço: 119.90
  },
  {
    id: 3,
    nome: "Boneca Luna",
    imagem: "img/luna.jpg",
    descrição: "Boneca com vestido de princesa.",
    preço: 149.90
  },
  {
    id: 4,
    nome: "Boneca Valentina",
    imagem: "img/valentina.jpg",
    descrição: "Boneca ruiva com roupas modernas.",
    preço: 99.90
  },
  {
    id: 5,
    nome: "Boneca Bella",
    imagem: "img/bella.jpg",
    descrição: "Boneca delicada com conjunto de acessórios.",
    preço: 129.90
  }
];

const cart = [];

function adicionarAoCarrinho(id) {
  const product = products.find(p => p.id === id);
  if (product) {
    cart.push(product);
    console.log("Produto adicionado ao carrinho.");
  }
}

function mostrarProdutos() {
  const produtosHtml = products.map(p => `
    <div class="card">
      <img src="${p.imagem}" alt="${p.nome}">
      <h2>${p.nome}</h2>
      <p>${p.descrição}</p>
      <p>Preço: R$ ${p.preço}</p>
      <button onclick="adicionarAoCarrinho(${p.id})">Comprar</button>
    </div>
  `).join("");
  document.getElementById("produtos").innerHTML = produtosHtml;
}

function mostrarCarrinho() {
  const carrinhoHtml = cart.map(p => `
    <div>
      <h2>${p.nome}</h2>
      <p>Preço: R$ ${p.preço}</p>
    </div>
  `).join("");
  document.getElementById("carrinho").innerHTML = carrinhoHtml;
}

function mostrarProdutos() {
  mostrarProdutos();
  mostrarCarrinho();
}

mostrarProdutos();
```

**backend.py**

```python
from flask import Flask, request, jsonify
import sqlite3

app = Flask(__name__)

# Conexão com o banco de dados
conn = sqlite3.connect('database.db')
cursor = conn.cursor()

# Criação das tabelas
cursor.execute('''
  CREATE TABLE IF NOT EXISTS produtos (
    id INTEGER PRIMARY KEY,
    nome TEXT NOT NULL,
    imagem TEXT NOT NULL,
    descrição TEXT NOT NULL,
    preço REAL NOT NULL
  );
''')

# Insere os produtos
products = [
  ('Boneca Sofia', 'img/sofia.jpg', 'Boneca com vestido rosa e cabelos longos.', 89.90),
  ('Boneca Alice', 'img/alice.jpg', 'Boneca fashion com acessórios especiais.', 119.90),
  ('Boneca Luna', 'img/luna.jpg', 'Boneca com vestido de princesa.', 149.90),
  ('Boneca Valentina', 'img/valentina.jpg', 'Boneca ruiva com roupas modernas.', 99.90),
  ('Boneca Bella', 'img/bella.jpg', 'Boneca delicada com conjunto de acessórios.', 129.90)
]
cursor.executemany('INSERT INTO produtos VALUES (?, ?, ?, ?)', products)

# Fechamento da conexão
conn.commit()
conn.close()

@app.route('/')
def index():
  return 'Bem-vindo ao site Pink Dolls!'

if __name__ == '__main__':
  app.run(debug=True)
```
