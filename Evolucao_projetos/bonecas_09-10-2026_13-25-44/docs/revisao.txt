// Função para carregar os produtos no carrinho
function carregarProdutosCarrinho() {
  const cart = document.getElementById("carrinho");
  cart.innerHTML = "";

  fetch("/api/produtos")
    .then(response => response.json())
    .then(data => {
      const produtos = data;
      const card = document.createElement("div");
      card.classList.add("card");

      produtos.forEach(produto => {
        const cardContent = document.createElement("div");
        cardContent.classList.add("card-content");

        const produtoImg = document.createElement("img");
        produtoImg.src = produto.imagem;
        cardContent.appendChild(produtoImg);

        const produtoNome = document.createElement("h2");
        produtoNome.textContent = produto.nome;
        cardContent.appendChild(produtoNome);

        const produtoPreco = document.createElement("p");
        produtoPreco.textContent = `R$ ${produto.preço.toFixed(2)}`;
        cardContent.appendChild(produtoPreco);

        card.appendChild(cardContent);
      });

      cart.appendChild(card);
    })
    .catch(erro => {
      const cart = document.getElementById("carrinho");
      cart.innerHTML = "<p>Não foi possível carregar os produtos.</p>";
    });
}

// Função para adicionar um produto ao carrinho
function adicionarProdutoAoCarrinho(produtoId) {
  fetch(`/api/produtos/${produtoId}`, { method: "GET" })
    .then(response => response.json())
    .then(data => {
      const carrinho = document.getElementById("carrinho");
      const linha = data[0];
      const carta = document.createElement("div");
      carta.classList.add("carta");

      const imagem = document.createElement("img");
      imagem.src = linha.imagem;
      carta.appendChild(imagem);

      const nome = document.createElement("h2");
      nome.textContent = linha.nome;
      carta.appendChild(nome);

      const preco = document.createElement("p");
      preco.textContent = `R$ ${linha.preço.toFixed(2)}`;
      carta.appendChild(preco);

      carrinho.appendChild(carta);
    })
    .catch(erro => {
      const carrinho = document.getElementById("carrinho");
      carrinho.innerHTML = "<p>Não foi possível adicionar o produto ao carrinho.</p>";
    });
}

// Função para remover um produto do carrinho
function removerProdutoDoCarrinho(produtoId) {
  fetch(`/api/produtos/${produtoId}`, { method: "DELETE" })
    .then(response => response.json())
    .then(data => {
      const carrinho = document.getElementById("carrinho");
      const linha = data[0];
      const carta = document.querySelector(`.carta:contains(${linha.nome})`);
      if (carta) {
        carrinho.removeChild(carta);
      }
    })
    .catch(erro => {
      const carrinho = document.getElementById("carrinho");
      carrinho.innerHTML = "<p>Não foi possível remover o produto do carrinho.</p>";
    });
}

def listar_produtos():
  conn = get_db_connection()
  linhas = conn.execute("SELECT id, nome, imagem, descrição, preço FROM produtos").fetchall()
  conn.close()
  return jsonify([dict(linha) for linha in linhas])

@app.route('/api/produtos/<int:item_id>', methods=["GET"])
def obter_produto(item_id):
  conn = get_db_connection()
  linha = conn.execute("SELECT id, nome, imagem, descrição, preço FROM produtos WHERE id = ?", (item_id,)).fetchone()
  conn.close()
  if linha is None:
    return jsonify({"erro": "Produto não encontrado"}), 404
  return jsonify(dict(linha))

@app.route('/api/produtos', methods=["POST"])
def criar_produto():
  dados = request.get_json()
  conn = get_db_connection()
  cursor = conn.execute("INSERT INTO produtos (nome, imagem, descrição, preço) VALUES (?, ?, ?, ?)", (dados["nome"], dados.get("imagem"), dados["descrição"], dados["preço"]))
  conn.commit()
  novo_id = cursor.lastrowid
  conn.close()
  return jsonify({"id": novo_id}), 201

@app.route('/api/produtos/<int:item_id>', methods=["DELETE"])
def remover_produto(item_id):
  conn = get_db_connection()
  conn.execute("DELETE FROM produtos WHERE id = ?", (item_id,))
  conn.commit()
  conn.close()
  return jsonify({"mensagem": "Produto removido com sucesso"})

  conn = sqlite3.connect('database.db')
