**Sistema de Gerenciamento de Exercícios da Academia**

**Frontend (Client-Side)**

### **Componentes**

#### **ExercíciosList.js**
```jsx
import React from 'react';
import { useNavigate, useState } from 'react';
import axios from 'axios';

const ExercíciosList = () => {
  const [exercicios, setExercícios] = useState([]);
  const navigate = useNavigate();

  const fetchExercícios = async () => {
    const response = await axios.get('/api/exercícios');
    setExercícios(response.data);
  };

  return (
    <div>
      <h1>Lista de Exercícios</h1>
      <ul>
        {exercícios.map((exercicio) => (
          <li key={exercicio.id_exercicio}>
            <p>
              {exercicio.nome_exercicio} ({exercicio.tipo_exercicio})
            </p>
          </li>
        ))}
      </ul>
      <button onClick={() => fetchExercícios()}>Atualizar lista</button>
    </div>
  );
};

export default ExercíciosList;
```

#### **AutenticaçãoComponent.js**
```jsx
import React from 'react';
import axios from 'axios';

const AutenticaçãoComponent = () => {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const navigate = useNavigate();

  const handleSubmit = async (event) => {
    event.preventDefault();
    try {
      await axios.post('/api/authenticate', { username, password });
      alert('Autenticação realizada com sucesso!');
      navigate('/exercicios');
    } catch (error) {
      console.error(error);
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      <label>
        Usuário:
        <input type="text" value={username} onChange={(event) => setUsername(event.target.value)} />
      </label>
      <br />
      <label>
        Senha:
        <input type="password" value={password} onChange={(event) => setPassword(event.target.value)} />
      </label>
      <button>Autenticar</button>
    </form>
  );
};

export default AutenticaçãoComponent;
```

#### **Serviços**

*   `services/exercicios.js`: Fornece métodos para criar, atualizar e remover os exercícios.
```js
import axios from 'axios';
import { Exercicio } from './models/exercicios';

const ExerciciosService = {
  create: async (exercicio) => {
    const response = await axios.post('/api/exercícios', ejercicio);
    return response.data;
  },
  update: async (idExercicio, exercicio) => {
    const response = await axios.put(`/api/exercícios/${idExercicio}`, exercicio);
    return response.data;
  },
  remove: async (idExercicio) => {
    try {
      await axios.delete(`/api/exercícios/${idExercicio}`);
      alert('Exercício excluído com sucesso!');
    } catch (error) {
      console.error(error);
    }
  },
};

export default ExerciciosService;
```

#### **Controleadores**

*   `controllers/exercicios.js`: Fornece métodos para lidar com as requisições relacionadas a exercícios.
```js
import { ExerciciosService } from './services/exercicios';

const ExerciciosController = {
  criar: async (req, res) => {
    const exercicio = req.body;
    try {
      const response = await ExerciciosService.create(exercicio);
      return res.status(201).json(response);
    } catch (error) {
      console.error(error);
      return res.status(500).json({ message: 'Erro ao criar o exercício' });
    }
  },
  atualizar: async (req, res) => {
    const idExercicio = req.params.idExercicio;
    const exercicio = req.body;
    try {
      const response = await ExerciciosService.update(idExercicio, exercicio);
      return res.status(200).json(response);
    } catch (error) {
      console.error(error);
      return res.status(500).json({ message: 'Erro ao atualizar o exercício' });
    }
  },
};

export default ExerciciosController;
```

#### **Roteadores**

*   `routes/exercícios.js`: Define os roteadores para as requisições relacionadas a exercícios.
```js
import express from 'express';
import { ExerciciosController } from './controllers/exercicios';

const router = express.Router();

router.post('/', ExerciciosController.criar);
router.put('/:idExercicio', ExerciciosController.atualizar);

export default router;
```
