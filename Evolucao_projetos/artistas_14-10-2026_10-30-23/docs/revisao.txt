```python
from flask import Flask, jsonify, request
from flask_sqlalchemy import SQLAlchemy
from flask_login import LoginManager

app = Flask(__name__)
app.config["SQLALCHEMY_DATABASE_URI"] = "sqlite:///database.db"
db = SQLAlchemy(app)
login_manager = LoginManager(app)

class Produto(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    nome = db.Column(db.String(100), nullable=False)
    descricao = db.Column(db.String(200), nullable=False)
    quantidade = db.Column(db.Integer, nullable=False)
    preco = db.Column(db.Float, nullable=False)

class Pedido(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    produto_id = db.Column(db.Integer, db.ForeignKey("produto.id"), nullable=False)
    quantidade = db.Column(db.Integer, nullable=False)

@login_manager.user_loader
def load_user(user_id):
    return User.query.get(int(user_id))

@app.route("/api/produtos", methods=["GET"])
def listar_produtos():
    produtos = Produto.query.all()
    return jsonify([{"id": produto.id, "nome": produto.nome, "descricao": produto.descricao, "quantidade": produto.quantidade, "preco": produto.preco} for produto in produtos])

@app.route("/api/produtos/<int:item_id>", methods=["GET"])
def obter_produto(item_id):
    produto = Produto.query.get(item_id)
    if produto is None:
        return jsonify({"erro": "Produto não encontrado"}), 404
    return jsonify({"id": produto.id, "nome": produto.nome, "descricao": produto.descricao, "quantidade": produto.quantidade, "preco": produto.preco})

@app.route("/carrinho", methods=["POST"])
def atualizar_carrinho():
    dados = request.get_json()
    produto = Produto.query.get(dados["produtoId"])
    if produto is None:
        return jsonify({"erro": "Produto não encontrado"}), 404
    pedido = Pedido.query.get(dados["id"])
    if pedido is None:
        return jsonify({"erro": "Pedido não encontrado"}), 404
    pedido.quantidade += dados["quantidade"]
    db.session.commit()
    return jsonify({"mensagem": "Pedido atualizado com sucesso"}), 200

@app.route("/pedidos", methods=["GET"])
def listar_pedidos():
    pedidos = Pedido.query.all()
    return jsonify([{"id": pedido.id, "produto_id": pedido.produto_id, "quantidade": pedido.quantidade} for pedido in pedidos])
```
