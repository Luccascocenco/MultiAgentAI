    ```python
# db.py
import mysql.connector

class Db:
    def __init__(self, host, database, user, password):
        self.host = host
        self.database = database
        self.user = user
        self.password = password
        self.conn = None

    def conectar(self):
        self.conn = mysql.connector.connect(
            host=self.host,
            database=self.database,
            user=self.user,
            password=self.password
        )

    def get_cursor(self):
        return self.conn.cursor()

# AlimentosService.py
from db import Db

class AlimentosService:
    def __init__(self, db):
        self.db = db

    def inserir_alimento(self, nome, preco):
        cursor = self.db.get_cursor()
        query = "INSERT INTO alimentos (nome, preco) VALUES (%s, %s)"
        values = (nome, preco)
        cursor.execute(query, values)
        self.db.conn.commit()
        return cursor.lastrowid

# main.py
from AlimentosService import AlimentosService

if __name__ == "__main__":
    db = Db('localhost', 'hamburgesa', 'admin', 'senhas')
    db.conectar()

    service = AlimentosService(db)
    nome = input("Digite o nome do alimento: ")
    preco = float(input("Digite o preço do alimento: "))

    id_alimento = service.inserir_alimento(nome, preco)

    print(f"Alimento inserido com sucesso! O ID é {id_alimento}")
```
Neste exemplo, estamos criando um serviço de inserção de alimentos que utiliza a biblioteca `mysql-connector-python` para se conectar ao banco de dados MySQL. O serviço tem um método `inserir_alimento` que recebe o nome e preço do alimento e insere-os no banco de dados.
