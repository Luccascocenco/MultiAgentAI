**Exemplo de código:**

Aqui está um exemplo de código em JavaScript para criar uma API de compras:
```javascript
const express = require('express');
const app = express();
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');

const users = [
  { id: 1, nome: 'João', senha: '123456' },
  { id: 2, nome: 'Maria', senha: '654321' },
];

app.get('/compras', (req, res) => {
  res.json(users);
});

app.post('/compras', (req, res) => {
  const { nome, senha } = req.body;
  const user = users.find((user) => user.nome === nome);
  if (!user) {
    return res.status(404).send({ mensagem: 'Usuário não encontrado' });
  }
  bcrypt.compare(senha, user.senha, (err, isValid) => {
    if (!isValid) {
      return res.status(401).send({ mensagem: 'Senha incorreta' });
    }
    const token = jwt.sign({ userId: user.id }, process.env.SECRET_KEY, { expiresIn: '1h' });
    res.send({ token });
  });
});

app.listen(3000, () => {
  console.log('Servidor rodando na porta 3000');
});
```
