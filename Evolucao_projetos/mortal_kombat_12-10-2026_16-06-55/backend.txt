**Backend**

```python
from flask import Flask, jsonify, request
import sqlite3
import os

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
DB_PATH = os.path.join(BASE_DIR, "database.db")
SCHEMA_PATH = os.path.join(BASE_DIR, "database", "schema.sql")

app = Flask(__name__, static_folder="static", static_url_path="")

def get_db_connection():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def inicializar_banco():
    # Cria o banco a partir do schema.sql, só na primeira vez
    if os.path.exists(DB_PATH) and os.path.getsize(DB_PATH) > 0:
        return

    conn = get_db_connection()
    with open(SCHEMA_PATH, encoding="utf-8") as arquivo:
        conn.executescript(arquivo.read())
    conn.commit()
    conn.close()

@app.route("/")
def home():
    return app.send_static_file("index.html")

@app.route("/login", methods=["POST"])
def login():
    dados = request.get_json()
    conn = get_db_connection()
    cursor = conn.execute(
        "SELECT * FROM usuarios WHERE email = ? AND senha = ?", (dados["email"], dados["senha"])
    )
    linha = cursor.fetchone()
    conn.close()
    if linha is None:
        return jsonify({"erro": "Credenciais inválidas"}), 401
    return jsonify({"id": linha[0]}), 200

@app.route("/api/usuarios", methods=["GET"])
def listar_usuarios():
    conn = get_db_connection()
    linhas = conn.execute("SELECT * FROM usuarios").fetchall()
    conn.close()
    return jsonify([dict(linha) for linha in linhas])

@app.route("/api/jogos", methods=["GET"])
def listar_jogos():
    conn = get_db_connection()
    linhas = conn.execute("SELECT * FROM jogos").fetchall()
    conn.close()
    return jsonify([dict(linha) for linha in linhas])

@app.route("/api/carrinhos", methods=["POST"])
def adicionar_jogo_carrinho():
    dados = request.get_json()
    conn = get_db_connection()
    cursor = conn.execute(
        "INSERT INTO carrinhos (usuario_id, jogo_id, quantidade) VALUES (?, ?, ?)",
        (dados["usuario_id"], dados["jogo_id"], dados["quantidade"])
    )
    conn.commit()
    conn.close()
    return jsonify({" mensagem": "Jogo adicionado ao carrinho"})

@app.route("/api/compras", methods=["POST"])
def realizar_compra():
    dados = request.get_json()
    conn = get_db_connection()
    cursor = conn.execute(
        "INSERT INTO compras (usuario_id, jogo_id, quantidade, preco, data) VALUES (?, ?, ?, ?, ?)",
        (dados["usuario_id"], dados["jogo_id"], dados["quantidade"], dados["preco"], dados["data"])
    )
    conn.commit()
    conn.close()
    return jsonify({" mensagem": "Compra realizada"})

if __name__ == "__main__":
    inicializar_banco()
    porta = int(os.environ.get("PORT", 5000))
    app.run(debug=False, port=porta)
```

function adicionarJogoAoCarrinho() {
    const jogo = document.getElementById("jogoEscolhido").value;
    const quantidade = document.getElementById("quantidade").value;

        localStorage.setItem("carrinho", JSON.stringify(carrinho));
        document.getElementById("carrinho").innerHTML = "";

            const lista = document.getElementById("carrinho");
            const li = document.createElement("li");
            li.textContent = `${item.quantidade} x ${item.jogo}`;

    }
}
