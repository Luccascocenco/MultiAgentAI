**Modelos de Dados**

```python
from flask_sqlalchemy import SQLAlchemy
from sqlalchemy import Integer, String, Float, Date, ForeignKey, Column, Enum
from sqlalchemy.orm import relationship

db = SQLAlchemy()

class Produto(db.Model):
    id_produto = db.Column(Integer, primary_key=True)
    nome = db.Column(String, nullable=False)
    preco_produto = db.Column(Float, nullable=False)
    avaliacao_produto = db.Column(Enum('Excelente', 'Bom', 'Ruim', name='avaliacao_produto'))
    fotos = db.relationship('Foto', backref='produto', lazy=True)
    pedidos = relationship('Pedido', backref='produto', lazy=True)
    favoritos = relationship('Favorito', backref='produto', lazy=True)

class Pedido(db.Model):
    id_pedido = db.Column(Integer, primary_key=True)
    id_produto = db.Column(Integer, ForeignKey('produto.id_produto'), nullable=False)
    quantidade = db.Column(Integer, nullable=False)
    subtotal = db.Column(Float, nullable=False)
    data_pedido = db.Column(Date, nullable=False)

class Favorito(db.Model):
    id_favorito = db.Column(Integer, primary_key=True)
    id_produto = db.Column(Integer, ForeignKey('produto.id_produto'), nullable=False)
    data_favorito = db.Column(Date, nullable=False)
```

**Banco de Dados**

```python
from flask import Flask, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///db.db'
db = SQLAlchemy(app)

@app.route('/api/produtos', methods=['GET'])
def get_produtos():
    produtos = Produto.query.all()
    return jsonify([dict(produto) for produto in produtos])

def get_produto(item_id):
    produto = Produto.query.get(item_id)
    return jsonify(dict(produto) if produto else {'erro': 'Produto não encontrado'})
```
