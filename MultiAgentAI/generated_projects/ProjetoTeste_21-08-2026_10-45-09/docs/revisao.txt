**Tabelas no banco de dados PostgreSQL:**

```sql
CREATE TABLE produtos (
  id SERIAL PRIMARY KEY,
  titulo VARCHAR(255) NOT NULL,
  autor VARCHAR(255) NOT NULL,
  categoria VARCHAR(255) NOT NULL,
  tipo VARCHAR(255) NOT NULL,
  preco DECIMAL(10, 2) NOT NULL,
  quantidade_em_estoque INTEGER NOT NULL
);

CREATE TABLE livros (
  id SERIAL PRIMARY KEY,
  titulo VARCHAR(255) NOT NULL,
  autor VARCHAR(255) NOT NULL,
  categoria VARCHAR(255) NOT NULL,
  tipo VARCHAR(255) NOT NULL,
  preco DECIMAL(10, 2) NOT NULL,
  quantidade_em_estoque INTEGER NOT NULL
);

CREATE TABLE ebooks (
  id SERIAL PRIMARY KEY,
  titulo VARCHAR(255) NOT NULL,
  autor VARCHAR(255) NOT NULL,
  categoria VARCHAR(255) NOT NULL,
  tipo VARCHAR(255) NOT NULL,
  preco DECIMAL(10, 2) NOT NULL,
  formato_ebook VARCHAR(255) NOT NULL,
  numero_de_paginas INTEGER NOT NULL,
  idioma VARCHAR(255) NOT NULL,
  editora VARCHAR(255) NOT NULL,
  ano_de_publicacao INTEGER NOT NULL
);

CREATE TABLE promo (
  id SERIAL PRIMARY KEY,
  tipo_promo VARCHAR(255) NOT NULL,
  valor_promo DECIMAL(10, 2) NOT NULL,
  produto INTEGER NOT NULL
);

CREATE TABLE avaliacoes (
  id SERIAL PRIMARY KEY,
  avaliacao FLOAT NOT NULL,
  comentario VARCHAR(255) NOT NULL,
  produto INTEGER NOT NULL
);

CREATE TABLE pedidos (
  id SERIAL PRIMARY KEY,
  produto INTEGER NOT NULL,
  quantidade INTEGER NOT NULL,
  subtotal DECIMAL(10, 2) NOT NULL,
  desconto DECIMAL(10, 2) NOT NULL,
  valor_total DECIMAL(10, 2) NOT NULL
);

CREATE TABLE carrinhos (
  id SERIAL PRIMARY KEY,
  produto INTEGER NOT NULL,
  quantidade INTEGER NOT NULL
);

CREATE TABLE capas (
  id SERIAL PRIMARY KEY,
  produto INTEGER NOT NULL,
  capa VARCHAR(255) NOT NULL
);

CREATE TABLE registros (
  id SERIAL PRIMARY KEY,
  produto INTEGER NOT NULL,
  data_cadastro DATE NOT NULL,
  ultima_atualizacao DATE NOT NULL
);
```
