**database.sql**
```sql
CREATE TABLE Alimentos (
  id INT PRIMARY KEY AUTO_INCREMENT,
  nome VARCHAR(255) NOT NULL,
  preco DECIMAL(10,2) NOT NULL,
  tipo ENUM('hamburgueres', 'saladas', 'outros') DEFAULT 'outros'
);

CREATE TABLE Pedidos (
  id INT PRIMARY KEY AUTO_INCREMENT,
  data DATE NOT NULL,
  hora TIME NOT NULL,
  total DECIMAL(10,2) NOT NULL,
  FOREIGN KEY (id_alimento) REFERENCES Alimentos(id)
);

CREATE TABLE Estoque (
  id INT PRIMARY KEY AUTO_INCREMENT,
  id_alimento INT,
  quantidade INT NOT NULL DEFAULT 0,
  FOREIGN KEY (id_alimento) REFERENCES Alimentos(id)
);

CREATE INDEX idx_alimentos_nome ON Alimentos (nome);
CREATE INDEX idx_pedidos_data ON Pedidos (data);
CREATE INDEX idx_estoque_id_alimento ON Estoque (id_alimento);

ALTER TABLE Pedidos
ADD COLUMN id_user INT;

ALTER TABLE Estoque
ADD COLUMN id_user INT;
```
Espero que isso atenda às suas necessidades! Lembre-se de que este é apenas o banco de dados, e você precisará desenvolver os módulos de inserção de alimentos, pedidos e gerenciamento de loja para completar a arquitetura do sistema.
