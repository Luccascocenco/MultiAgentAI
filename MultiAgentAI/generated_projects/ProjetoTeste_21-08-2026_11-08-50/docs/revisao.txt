```python
import requests

url = 'http://localhost:5000/produtos'
data = {
    'titulo': 'New Product',
    'autor': 'John Doe',
    'categoria': 'Electronics',
    'tipo': 'Product',
    'preco': 99.99,
    'preco_promocional': 49.99,

    'ano_publicacao': 2020,
    'avaliacao': 4.5,
    'quantidade_avalies': 10,
    'status': 'Active',
    'data_cadastro': '2022-01-01',
    'url_capa': 'https://example.com/capa.jpg',
    'palavras_chave': 'product,electronics,book'
}

response = requests.post(url, json=data)

if response.status_code == 201:
    print('Product created successfully!')
else:
    print('Error creating product:', response.text)
```
This code sends a `POST` request to the `/produtos` route with a JSON payload containing the new product's data. If the request is successful, it prints a success message. Otherwise, it prints an error message with the response text.
