**Arquitetura**

*   **Serviços**: são responsáveis por realizar operações específicas, como criar um novo usuário ou exercício.
*   **Controleadores**: lidam com as requisições HTTP e chamam os serviços relevantes para realizar a lógica de negócios.
*   **Módulos**: são conjuntos de controladores que trabalham juntos para lidar com uma parte específica do aplicativo.

**Backend**

```javascript
// services/exerciciosService.js
const mongoose = require('mongoose');

const ExerciciosSchema = new mongoose.Schema({
  nomeExercicio: String,
  tipoExercicio: String,
  tempoDuração: Number,
  dificuldade: String,
  numeroRepeticoes: Number,
});

const ExerciciosService = {
  criarExercicio(exercicio) {
    const novoExercicio = new ExerciciosSchema({
      nomeExercicio,
      tipoExercicio,
      tempoDuração,
      dificuldade,
      numeroRepeticoes,
    });

    return novoExercicio.save();
  },
};

module.exports = ExerciciosService;
```

```javascript
// controllers/exerciciosController.js
const express = require('express');
const router = express.Router();
const ExerciciosService = require('../services/exerciciosService');

router.post('/exercicios', async (req, res) => {
  const { nomeExercicio, tipoExercicio, tempoDuração, dificuldade, numeroRepeticoes } = req.body;
  try {
    const novoExercicio = await ExerciciosService.criarExercicio({
      nomeExercicio,
      tipoExercicio,
      tempoDuração,
      dificuldade,
      numeroRepeticoes,
    });
    res.status(201).json(novoExercicio);
  } catch (error) {
    console.error(error);
    res.status(500).json({ message: 'Erro ao criar exercício' });
  }
});

router.get('/exercicios', async (req, res) => {
  const Exercicios = await Exercicios.find().exec();
  res.json(Exercicios);
});

module.exports = router;
```

```javascript
// components/ExerciciosList.js
import React from 'react';

const ExerciciosList = () => {
  const [exercicios, setExercicios] = React.useState([]);

  React.useEffect(() => {
    fetch('/exercicios')
      .then((res) => res.json())
      .then((data) => setExercicios(data));
  }, []);

  return (
    <div>
      <h1>Lista de Exercícios</h1>
      <ul>
        {exercicios.map((exercicio) => (
          <li key={exercicio._id}>
            {exercicio.nomeExercicio}
          </li>
        ))}
      </ul>
    </div>
  );
};

export default ExerciciosList;
```

**Banco de Dados**

```sql
CREATE TABLE usuarios (
  id_user INT PRIMARY KEY AUTO_INCREMENT,
  nome VARCHAR(255) NOT NULL,
  email VARCHAR(255) NOT NULL UNIQUE,
  senha VARCHAR(255) NOT NULL
);

CREATE TABLE exercicios (
  id_exercicio INT PRIMARY KEY AUTO_INCREMENT,
  nome_exercicio VARCHAR(255) NOT NULL,
  tipo_exercicio ENUM('com equipamento', 'sem equipamento') NOT NULL,
  tempo_duração INT NOT NULL,
  dificuldade ENUM('fácil', 'medio', 'dificil') NOT NULL,
  numero_repeticoes INT NOT NULL
);

CREATE TABLE logotransacoes (
  id_logotran INT PRIMARY KEY AUTO_INCREMENT,
  id_user INT NOT NULL,
  id_exercicio INT NOT NULL,
  data_transacao DATE NOT NULL,
  hora_transacao TIME NOT NULL,
  FOREIGN KEY (id_user) REFERENCES usuarios(id_user),
  FOREIGN KEY (id_exercicio) REFERENCES exercicios(id_exercicio)
);
```

**Observações:**

*   A tabela `usuarios` armazena informações sobre os usuários, incluindo o nome, e-mail e senha.
*   A tabela `exercicios` armazena informações sobre os exercícios, incluindo o nome do exercício, tipo de exercício, tempo de duração, dificuldade e número de repetições.
*   A tabela `logotransacoes` armazena registros das transações realizadas pelos usuários, incluindo a data e hora da transação, bem como as informações sobre o usuário e o exercício envolvido.
