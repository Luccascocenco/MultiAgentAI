Aqui está um exemplo de como você pode implementar essa arquitetura em Node.js e Express:

**Backend**

*   **Serviços**
    ```javascript
// services/exerciciosService.js
const mongoose = require('mongoose');

const ExerciciosSchema = new mongoose.Schema({
  nomeExercicio: String,
  tipoExercicio: String,
  tempoDuração: Number,
  dificuldade: String,
  numeroRepeticoes: Number,
});

const ExerciciosService = {
  criarExercicio(exercicio) {
    const novoExercicio = new ExerciciosSchema({
      nomeExercicio,
      tipoExercicio,
      tempoDuração,
      dificuldade,
      numeroRepeticoes,
    });

    return novoExercicio.save();
  },
  atualizarExercicio(id, exercicioAtualizado) {
    const Exercicio = await Exercicios.find({ _id: id });
    if (Exercicio) {
      Exercicios.updateOne(
        { _id: id },
        { $set: { nomeExercicio: exercicioAtualizado.nomeExercicio } }
      );
    }
  },
};

module.exports = ExerciciosService;
```

*   **Controleadores**
    ```javascript
// controllers/exerciciosController.js
const express = require('express');
const router = express.Router();
const ExerciciosService = require('../services/exerciciosService');

router.post('/exercicios', async (req, res) => {
  const { nomeExercicio, tipoExercicio, tempoDuração, dificuldade, numeroRepeticoes } = req.body;
  try {
    const novoExercicio = await ExerciciosService.criarExercicio({
      nomeExercicio,
      tipoExercicio,
      tempoDuração,
      dificuldade,
      numeroRepeticoes,
    });
    res.status(201).json(novoExercicio);
  } catch (error) {
    console.error(error);
    res.status(500).json({ message: 'Erro ao criar exercício' });
  }
});

router.get('/exercicios', async (req, res) => {
  const Exercicios = await Exercicios.find().exec();
  res.json(Exercicios);
});

module.exports = router;
```

*   **Módulos**
    ```javascript
// modules/exerciciosModule.js
const express = require('express');
const router = express.Router();
const ExerciciosController = require('../controllers/exerciciosController');

router.post('/exercicios', ExerciciosController.criarExercicio);
router.get('/exercicios', ExerciciosController.exerciciosList);

module.exports = router;
```

**Frontend**

*   **Componentes**
    ```javascript
// components/ExerciciosList.js
import React from 'react';

const ExerciciosList = () => {
  const [exercicios, setExercicios] = React.useState([]);

  React.useEffect(() => {
    fetch('/exercicios')
      .then((res) => res.json())
      .then((data) => setExercicios(data));
  }, []);

  return (
    <div>
      <h1>Lista de Exercícios</h1>
      <ul>
        {exercicios.map((exercicio) => (
          <li key={exercicio._id}>
            {exercicio.nomeExercicio}
          </li>
        ))}
      </ul>
    </div>
  );
};

export default ExerciciosList;
```
