**Backend**

```python
from flask import Flask, jsonify, request
import sqlite3
import os

# Configurações do Flask
app = Flask(__name__, static_folder="static", static_url_path="")


# Conexão com banco de dados
def get_db_connection():
    conn = sqlite3.connect("database.db")
    conn.row_factory = sqlite3.Row
    return conn


# Inicialização do banco de dados
def inicializar_banco():
    conn = get_db_connection()
    with open("database/schema.sql", encoding="utf-8") as arquivo:
        conn.executescript(arquivo.read())
    conn.commit()
    conn.close()


# Rota para listar produtos
@app.route("/api/produtos", methods=["GET"])
def listar_produtos():
    conn = get_db_connection()
    produtos = conn.execute("SELECT id, nome, descricao, preco FROM produtos ORDER BY preco ASC").fetchall()
    conn.close()
    return jsonify([dict(linha) for linha in produtos])


# Rota para criar um novo produto
@app.route("/api/produtos", methods=["POST"])
def criar_produto():
    dados = request.get_json()
    conn = get_db_connection()
    conn.execute("INSERT INTO produtos (nome, descricao, preco) VALUES (?, ?, ?)", (dados["nome"], dados.get("descricao"), dados["preco"]))
    conn.commit()
    produto_id = conn.lastrowid
    conn.close()
    return jsonify({"id": produto_id}), 201


# Rota para atualizar pedido
@app.route("/carrinho", methods=["POST"])
def atualizar_carrinho():
    dados = request.get_json()
    conn = get_db_connection()
    conn.execute("UPDATE pedidos SET quantidade = ? WHERE id = ?", (dados["quantidade"], dados["produto_id"]))
    conn.commit()
    conn.close()
    return jsonify({"mensagem": "Produto atualizado com sucesso"})


# Rota para visualizar pedido
@app.route("/pedidos", methods=["GET"])
def visualizar_pedidos():
    conn = get_db_connection()
    pedidos = conn.execute("SELECT id, produtoId, quantidade FROM pedidos").fetchall()
    conn.close()
    return jsonify([dict(linha) for linha in pedidos])


# Rota para consultar produto por ID
@app.route("/api/produtos/<int:produto_id>", methods=["GET"])
def consultar_produto(produto_id):
    conn = get_db_connection()
    produto = conn.execute("SELECT nome, descricao, preco FROM produtos WHERE id = ?", (produto_id,)).fetchone()
    conn.close()
    if produto is None:
        return jsonify({"erro": "Produto não encontrado"}), 404
    return jsonify({"nome": produto[0], "descricao": produto[1], "preco": produto[2]})


# Inicialização do servidor
def inicializar_servidor():
    inicializar_banco()
    porta = int(os.environ.get("PORT", 5000))
    app.run(debug=False, port=porta)


if __name__ == "__main__":
    inicializar_servidor()
```

* Arquivo `script.js`:

```javascript
// Carregar produtos
fetch('/api/produtos')
    .then(response => response.json())
    .then(produtos => {
        produtos.forEach(produto => {
            const produtoElemento = document.createElement('tr');
            produtoElemento.innerHTML = `
                <td>${produto.nome}</td>
                <td>${produto.descricao}</td>
                <td>${produto.preco}</td>
                <td><a href="#" class="action">Adicionar ao Carrinho</a></td>
            `;
            document.getElementById('produtos-tabela').appendChild(produtoElemento);
        });
    });

// Adicionar produto ao carrinho
document.querySelectorAll('.action').forEach(actionElement => {
    actionElement.addEventListener('click', () => {
        fetch('/carrinho', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({ produtoId: actionElement.dataset.produtoId, quantidade: 1 })
        })
            .then(response => response.json())
            .then(mensagem => console.log(mensagem));
    });
});

// Mostrar pedido atual
fetch('/carrinho')
    .then(response => response.json())
    .then(pedido => {
        pedido.forEach(produto => {
            const produtoElemento = document.createElement('tr');
            produtoElemento.innerHTML = `
                <td>${produto.nome}</td>
                <td>${produto.quantidade}</td>
                <td><a href="#" class="action">Atualizar Quantidade</a></td>
            `;
            document.getElementById('carrinho-tabela').appendChild(produtoElemento);
        });
    });

// Atualizar quantidade de produto no carrinho
document.querySelectorAll('.action').forEach(actionElement => {
    actionElement.addEventListener('click', () => {
        fetch('/carrinho', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            },

        })
            .then(response => response.json())
            .then(mensagem => console.log(mensagem));
    });
});
```
